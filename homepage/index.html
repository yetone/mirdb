Quick Start

Get started with MirDB using your favorite Memcached client. MirDB is fully compatible with the Memcached protocol.

Python 
from pymemcache.client.base import Client

# Connect to MirDB on localhost:12333
client = Client(('localhost', 12333))

# Set a key-value pair
client.set('greeting', 'Hello, MirDB!')

# Retrieve the value
result = client.get('greeting')
print(result)  # b'Hello, MirDB!'
Go 
package main

import (
    "github.com/bradfitz/gomemcache/memcache"
    "log"
)

func main() {
    // Connect to MirDB on localhost:12333
    mc := memcache.New("localhost:12333")

    // Set a key-value pair
    mc.Set(&memcache.Item{Key: "greeting", Value: []byte("Hello, MirDB!")})

    // Retrieve the value
    item, err := mc.Get("greeting")
    if err != nil {
        log.Fatal(err)
    }
    log.Printf("%s", item.Value)
}
